import json,os,subprocess,tempfile
from http.server import BaseHTTPRequestHandler,ThreadingHTTPServer
from pathlib import Path
from threading import Thread
root=Path.cwd()
assert '33306b0' in (root/'evolution-06/native-reassessment/flight-card.md').read_text()
scratch=Path(tempfile.mkdtemp(prefix='e6-native-'))
print(scratch,flush=True)
marker='E6-WITNESS-VERIFY'
requests=[]
class Handler(BaseHTTPRequestHandler):
 def log_message(self,*args): pass
 def do_POST(self):
  data=json.loads(self.rfile.read(int(self.headers['Content-Length'])))
  messages=data['messages']
  text=json.dumps(messages)
  names=[x.get('function',{}).get('name') for x in data.get('tools',[])]
  continuation='Continue if you have next steps' in text
  if continuation:
   stage,content,count='continuation','Next action: '+marker,100
  elif 'read' in names and marker in text:
   stage,count='initial',6500
   content=None
  elif marker in text:
   stage,content,count='compaction_or_title','## Next action\n'+marker,100
  else:
   stage,content,count='setup','Course witness',100
  requests.append(dict(stage=stage,has_retained_action=marker in text,read_exposed='read' in names,scripted_input=count))
  if len(requests)>12:
   self.send_error(400);return
  self.send_response(200)
  self.send_header('Content-Type','text/event-stream')
  self.end_headers()
  reason='stop'
  delta={'content':content}
  if content is None:
   reason='tool_calls'
   delta={'tool_calls':[{'index':0,'id':'e6-read-fixture','type':'function',
        'function':{'name':'read','arguments':json.dumps({'filePath':str(scratch/'fixture.txt')})}}]}
  for d,r,u in ((delta,None,None),({},reason,None),({},None,dict(prompt_tokens=count,completion_tokens=10,total_tokens=count+10))):
   packet=dict(id='e6-witness',object='chat.completion.chunk',created=0,model='boundary')
   packet['choices']=[] if u else [dict(index=0,delta=d,finish_reason=r)]
   if u: packet['usage']=u
   self.wfile.write(('data: '+json.dumps(packet)+'\n\n').encode())
  self.wfile.write(b'data: [DONE]\n\n')
server=ThreadingHTTPServer(('127.0.0.1',0),Handler)
Thread(target=server.serve_forever,daemon=True).start()
(scratch/'fixture.txt').write_text('Synthetic read-only fixture.\n')
config={
 'share':'disabled','model':'local/boundary','enabled_providers':['local'],
 'compaction':{'auto':True,'reserved':0},
 'provider':{'local':{'npm':'@ai-sdk/openai-compatible','name':'Loopback',
 'options':{'baseURL':f'http://127.0.0.1:{server.server_port}/v1'},
 'models':{'boundary':{'name':'Boundary','limit':{'context':8000,'input':6400,'output':1000}}}}},
 'permission':{'*':'deny','read':'allow','external_directory':'deny'},
}
env={
 'PATH':os.environ['PATH'],'OPENCODE_DISABLE_MODELS_FETCH':'1','OPENCODE_DISABLE_AUTOUPDATE':'1',
 **{f'XDG_{k}_HOME':str(scratch/k.lower()) for k in ('CONFIG','DATA','CACHE','STATE')},
 'OPENCODE_CONFIG_CONTENT':json.dumps(config),
}
cmd=['opencode','run','--pure','--format','json','--model','local/boundary','--title','E6 native compaction',
    f'Read fixture.txt. Preserve the next action {marker} for the following turn.']
p=subprocess.run(cmd,cwd=scratch,env=env,capture_output=True,text=True,timeout=100)
(scratch/'raw.jsonl').write_text(p.stdout)
(scratch/'raw.stderr').write_text(p.stderr)
events=[json.loads(x) for x in p.stdout.splitlines() if x.startswith('{')]
sid=next(x['sessionID'] for x in events if 'sessionID' in x)
q=subprocess.run(['opencode','export',sid,'--sanitize','--pure'],cwd=scratch,env=env,capture_output=True,text=True,check=True,timeout=30)
export=json.loads(q.stdout)
message_rows=[]
for msg in export['messages']:
 info=msg['info']
 parts=[{'type':x['type'],**({'auto':x.get('auto')} if x['type']=='compaction' else {})}
        for x in msg['parts']]
 message_rows.append(dict(id=info['id'],session=info['sessionID'],role=info['role'],
                          summary=info.get('summary',False),tokens=info.get('tokens'),parts=parts))
result=dict(exit_status=p.returncode,session=sid,fixture_only=True,configuration=config['provider']['local']['models']['boundary']['limit'],
            reserve=0,execution='opencode run with no manual compact',requests=requests,messages=message_rows)
(scratch/'observed.json').write_text(json.dumps(result,indent=2)+'\n')
print(json.dumps(result,indent=2))
assert p.returncode==0
assert any(p['type']=='compaction' and p['auto'] is True for m in message_rows for p in m['parts'])
assert any(r['stage']=='continuation' and r['has_retained_action'] for r in requests)
